# PRoot for the Dextop Linux container feature (arm64-v8a only): termux/proot
# 4dba3afb with DroidDeck's patches (third_party/proot/droiddeck).
#
# proot and its loader are executables, named lib*.so so that the Android
# packager ships them in nativeLibraryDir (useLegacyPackaging extracts them).
# The privileged shell executes them from there; the app itself never does.

set(DEXTOP_THIRD_PARTY "${CMAKE_CURRENT_LIST_DIR}/../../../../../../third_party")
set(PROOT_SRC "${DEXTOP_THIRD_PARTY}/proot/src")
set(TALLOC_SRC "${DEXTOP_THIRD_PARTY}/talloc")
set(PROOT_GEN "${CMAKE_CURRENT_BINARY_DIR}/proot-generated")
file(MAKE_DIRECTORY "${PROOT_GEN}")

# Equivalent of proot's auto-generated build.h. Android 11+ (minSdk 30)
# always provides process_vm_readv/writev and seccomp filters.
file(WRITE "${PROOT_GEN}/build.h"
    "/* Generated by Dextop's CMakeLists.txt. */\n"
    "#ifndef BUILD_H\n#define BUILD_H\n"
    "#define VERSION \"termux-4dba3afb-droiddeck\"\n"
    "#define HAVE_PROCESS_VM\n#define HAVE_SECCOMP_FILTER\n"
    "#endif /* BUILD_H */\n")

# talloc, the only dependency of proot.
add_library(dextop_talloc STATIC "${TALLOC_SRC}/talloc.c")
target_include_directories(dextop_talloc PUBLIC "${TALLOC_SRC}")
target_compile_options(dextop_talloc PRIVATE -O2 -fPIC)

set(PROOT_DEFINES _FILE_OFFSET_BITS=64 _GNU_SOURCE ARG_MAX=131072
    PROOT_UNBUNDLE_LOADER="/data/local/tmp/dextop/linux/.loader")

# Loader: freestanding, static, at the fixed address from arch.h (arm64).
add_executable(proot_loader "${PROOT_SRC}/loader/loader.c" "${PROOT_SRC}/loader/assembly.S")
target_include_directories(proot_loader PRIVATE "${PROOT_GEN}" "${PROOT_SRC}")
target_compile_definitions(proot_loader PRIVATE ${PROOT_DEFINES})
target_compile_options(proot_loader PRIVATE -O2 -fPIC -ffreestanding)
# --no-gc-sections: proot jumps to pokedata_workaround by offset (loader-info.c),
# so nothing in the loader references it and the NDK's default
# --gc-sections would drop it.
target_link_options(proot_loader PRIVATE
    -static -nostdlib "LINKER:--build-id=none,-Ttext=0x2000000000,-z,noexecstack,--no-gc-sections")
set_target_properties(proot_loader PROPERTIES PREFIX "lib" OUTPUT_NAME "proot_loader" SUFFIX ".so")

find_program(DEXTOP_READELF NAMES llvm-readelf HINTS "${ANDROID_TOOLCHAIN_ROOT}/bin" REQUIRED NO_DEFAULT_PATH)
add_custom_command(
    OUTPUT "${PROOT_GEN}/loader-info.c"
    COMMAND "${CMAKE_COMMAND}" -DREADELF=${DEXTOP_READELF} -DLOADER=$<TARGET_FILE:proot_loader>
        -DOUTPUT=${PROOT_GEN}/loader-info.c -P "${CMAKE_CURRENT_LIST_DIR}/GenerateLoaderInfo.cmake"
    DEPENDS proot_loader "${CMAKE_CURRENT_LIST_DIR}/GenerateLoaderInfo.cmake"
    COMMENT "Generating proot loader-info.c"
    VERBATIM
)

# proot's object list (src/GNUmakefile), without the bundled-loader objects.
set(PROOT_SOURCES
    cli/cli.c cli/proot.c cli/note.c
    execve/enter.c execve/exit.c execve/shebang.c execve/elf.c execve/ldso.c execve/auxv.c execve/aoxp.c
    path/binding.c path/glue.c path/canon.c path/f2fs-bug.c path/path.c path/proc.c path/temp.c
    syscall/seccomp.c syscall/syscall.c syscall/chain.c syscall/enter.c syscall/exit.c syscall/sysnum.c
    syscall/socket.c syscall/heap.c syscall/rlimit.c
    tracee/tracee.c tracee/mem.c tracee/reg.c tracee/event.c tracee/seccomp.c tracee/statx.c
    ptrace/ptrace.c ptrace/user.c ptrace/wait.c
    extension/extension.c extension/ashmem_memfd/ashmem_memfd.c extension/kompat/kompat.c
    extension/fake_id0/chown.c extension/fake_id0/chroot.c extension/fake_id0/getsockopt.c
    extension/fake_id0/sendmsg.c extension/fake_id0/socket.c extension/fake_id0/open.c
    extension/fake_id0/unlink.c extension/fake_id0/rename.c extension/fake_id0/chmod.c
    extension/fake_id0/utimensat.c extension/fake_id0/access.c extension/fake_id0/exec.c
    extension/fake_id0/link.c extension/fake_id0/symlink.c extension/fake_id0/mk.c
    extension/fake_id0/stat.c extension/fake_id0/helper_functions.c extension/fake_id0/fake_id0.c
    extension/hidden_files/hidden_files.c extension/mountinfo/mountinfo.c
    extension/port_switch/port_switch.c extension/sysvipc/sysvipc.c extension/sysvipc/sysvipc_msg.c
    extension/sysvipc/sysvipc_sem.c extension/sysvipc/sysvipc_shm.c
    extension/link2symlink/link2symlink.c extension/fix_symlink_size/fix_symlink_size.c
)
list(TRANSFORM PROOT_SOURCES PREPEND "${PROOT_SRC}/")
# Relies on an implicit strcmp/memset declaration, which NDK 28 rejects.
set_source_files_properties("${PROOT_SRC}/extension/ashmem_memfd/ashmem_memfd.c"
    PROPERTIES COMPILE_OPTIONS "-include;string.h")

add_executable(proot ${PROOT_SOURCES} "${PROOT_GEN}/loader-info.c")
target_include_directories(proot PRIVATE "${PROOT_GEN}" "${PROOT_SRC}")
target_compile_definitions(proot PRIVATE ${PROOT_DEFINES})
target_compile_options(proot PRIVATE -O2 -Wall -Wextra -Wno-unused-parameter -Wno-sign-compare
    -Wno-error=implicit-function-declaration)
target_link_libraries(proot PRIVATE dextop_talloc)
target_link_options(proot PRIVATE "LINKER:-z,noexecstack")
set_target_properties(proot PROPERTIES PREFIX "lib" OUTPUT_NAME "proot" SUFFIX ".so")
